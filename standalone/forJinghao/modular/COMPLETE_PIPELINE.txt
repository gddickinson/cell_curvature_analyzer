================================================================================
🔧 CONFIGURATION OPTIONS:
================================================================================

Complete configuration example:

{
  "analysis": {
    "n_points": 12,              # Number of sampling points along edge
    "depth": 200,                # Sampling rectangle depth (pixels)
    "width": 75,                 # Sampling rectangle width (pixels)
    "min_cell_coverage": 0.8,    # Minimum coverage for valid sampling
    "try_rotation": true,        # Try 180° rotation for invalid rectangles
    "movement_threshold": 0.1,   # Movement classification threshold
    "min_movement_pixels": 5,    # Minimum valid displacements
    "exclude_endpoints": true,   # Exclude first/last edge points
    "temporal_direction": "past" # "past" or "future"
  },
  "output": {
    "save_detailed_csv": true,
    "save_summary_json": true,
    "save_correlation_plots": true,
    "save_frame_plots": true
  },
  "visualization": {
    "plot_every_nth_frame": 10,
    "movement_cmap": "RdBu_r",
    "intensity_cmap": "viridis",
    "figure_dpi": 150
  }
}
